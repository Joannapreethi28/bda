CREATE OR REPLACE TABLE faers.rx_flags AS
SELECT safetyreportid,
  COUNTIF(term = 'DRUG INEFFECTIVE') > 0 AS rx_drug_ineffective,
  COUNTIF(term = 'NAUSEA') > 0 AS rx_nausea,
  COUNTIF(term = 'FATIGUE') > 0 AS rx_fatigue,
  COUNTIF(term = 'OFF LABEL USE') > 0 AS rx_off_label_use,
  COUNTIF(term = 'DIARRHOEA') > 0 AS rx_diarrhoea,
  COUNTIF(term = 'PAIN') > 0 AS rx_pain,
  COUNTIF(term = 'HEADACHE') > 0 AS rx_headache,
  COUNTIF(term = 'DYSPNOEA') > 0 AS rx_dyspnoea,
  COUNTIF(term = 'DIZZINESS') > 0 AS rx_dizziness,
  COUNTIF(term = 'VOMITING') > 0 AS rx_vomiting,
  COUNTIF(term = 'RASH') > 0 AS rx_rash,
  COUNTIF(term = 'MALAISE') > 0 AS rx_malaise,
  COUNTIF(term = 'ARTHRALGIA') > 0 AS rx_arthralgia,
  COUNTIF(term = 'ASTHENIA') > 0 AS rx_asthenia,
  COUNTIF(term = 'PRURITUS') > 0 AS rx_pruritus,
  COUNTIF(term = 'INCORRECT DOSE ADMINISTERED') > 0 AS rx_incorrect_dose_administered,
  COUNTIF(term = 'PYREXIA') > 0 AS rx_pyrexia,
  COUNTIF(term = 'FALL') > 0 AS rx_fall,
  COUNTIF(term = 'INJECTION SITE PAIN') > 0 AS rx_injection_site_pain,
  COUNTIF(term = 'INSOMNIA') > 0 AS rx_insomnia
FROM (SELECT DISTINCT r.safetyreportid, UPPER(TRIM(x.reactionmeddrapt)) AS term
      FROM faers.drug_event_raw r, UNNEST(r.patient.reaction) x)
GROUP BY safetyreportid; -- at [2:1]
